  1. 主界面底部 HUD 状态栏 (#hud-bar) — 显示 HP/MP/耐力/契约者名称/生命状态
  2. 手机终端顶部状态栏 (.phone-status-bar) — 显示时间和连接状态

--- populate 函数：填充 HUD 数据 ---

function populate() {
  const d = D;   // D(path, def) = getNested(G.stat_data, path, def)

  // HP
  const hpm = d('契约者.衍生属性.HP_最大', 0);
  const hpc = d('契约者.衍生属性.HP_当前', 0);
  setText('hud-hp-num', hpc + '/' + hpm);
  setStyle('hud-hp-fill', 'width',
    Math.min(hpc / Math.max(hpm, 1) * 100, 100) + '%');

  // MP
  const mpm = d('契约者.衍生属性.MP_最大', 0);
  const mpc = d('契约者.衍生属性.MP_当前', 0);
  setText('hud-mp-num', mpc + '/' + mpm);
  setStyle('hud-mp-fill', 'width',
    Math.min(mpc / Math.max(mpm, 1) * 100, 100) + '%');

  // 耐力 (SP)
  const spm = d('契约者.衍生属性.耐力_最大', 0);
  const spc = d('契约者.衍生属性.耐力_当前', 0);
  setText('hud-sp-num', spc + '/' + spm);
  setStyle('hud-sp-fill', 'width',
    Math.min(spc / Math.max(spm, 1) * 100, 100) + '%');

  // 契约者名称
  setText('hud-name', d('契约者.头部.姓名', '---'));

  // 生命状态
  setText('hud-life-state', d('契约者.状态.生命状态', '健康'));
}

--- 辅助函数 ---

function setText(id, t) {
  const el = document.getElementById(id);
  if (el) el.textContent = t;
}

function setStyle(id, p, v) {
  const el = document.getElementById(id);
  if (el) el.style[p] = v;
}

function getNested(obj, path, def) {
  const keys = path.split('.');
  let cur = obj;
  for (const k of keys) {
    if (cur == null || typeof cur !== 'object') return def;
    cur = cur[k];
  }
  return cur !== undefined ? cur : def;
}

--- 初始化调用 ---

function init() {
  initEmbers();
  Overlay.init();
  populate();    // ← 填充状态栏
  bind();
}
document.addEventListener('DOMContentLoaded', init);

--- 更新时间 ---

function updatePhoneTime() {
  const now = new Date();
  const timeStr = now.getHours().toString().padStart(2, '0') + ':'
                + now.getMinutes().toString().padStart(2, '0');
  const dateStr = (now.getMonth() + 1) + '/' + now.getDate();
  const weekDays = ['日','一','二','三','四','五','六'];
  const weekStr = '周' + weekDays[now.getDay()];

  document.getElementById('ps-time').textContent = timeStr;
  document.getElementById('ph-time').textContent = timeStr;
  document.getElementById('ph-date').textContent = dateStr + ' ' + weekStr;

  const world = getGameData('当前世界', '');
  const gameDate = getGameData('当前时间.现实日期', '');
  const elWorld = document.getElementById('ph-world');
  if (elWorld) {
    const parts = [world, gameDate].filter(Boolean);
    elWorld.textContent = parts.length > 0
      ? parts.join(' · ') : '里世界时间流';
  }
}

═══════════════════════════════════════════════════════════════
  五、使用说明
═══════════════════════════════════════════════════════════════

1. 将 HTML 部分的 #hud-bar 放在 <body> 内合适位置
2. 将 CSS 变量和 HUD 样式引入你的样式表
3. 在 JS 中实现 populate() 函数，根据你的数据源更新状态栏
4. 状态栏响应式适配了 768px 和 480px 两种断点
5. 手机终端状态栏通过 Ctrl+M 快捷键呼出，支持拖动和置顶
