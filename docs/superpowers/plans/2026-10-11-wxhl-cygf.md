# wxhl-cygf 创意工坊宿主脚本 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为无限回廊角色卡实现创意工坊宿主脚本：注册「创意工坊」按钮 → 弹出云端工坊 iframe 面板 → 通过 postMessage 桥提供世界书操作原语（含 upsert），外加 Discord 登录转发和内容包格式契约（zod schema + 测试）。

**Architecture:** 瘦宿主（范例一协议思路）：安装编排在云端页面，宿主只执行桥接原语；消息监听挂 `window.parent`（范例二技术，无 script 注入）；独立世界书「无限回廊mod」挂角色卡附加世界书，无锚点条目；upsert 按 `extra.wxhlWorkshop.sourceId` 删旧插新。

**Tech Stack:** TypeScript + zod 4（`z` 自动导入）+ vitest（node 环境，显式 `import { describe, expect, it, vi } from 'vitest'`）+ webpack 打包为 `dist/wxhl-cygf/index.js`。

**Spec:** [docs/superpowers/specs/2026-10-11-wxhl-cygf-design.md](../specs/2026-10-11-wxhl-cygf-design.md)

- 宿主**不解析内容包**、**不写 MVU 变量**；「扮演/入队/挑战」是主卡 AI 的职责

- 脚本内 `$` 直接作用于酒馆主页面（`window.$ = window.parent.$`），无需手动找 parent document
- 常量（值逐字固定）：桥接 channel `wxhl-workshop:worldbook`；登录 channel `wxhl-workshop:auth`；世界书名 `无限回廊mod`；按钮名 `创意工坊`；默认工坊 URL `https://cygf.657868.xyz`；脚本变量覆盖键 `wxhl_cygf_workshop_url`；extra 把手键 `wxhlWorkshop`

// src/wxhl-cygf/bridge.ts
export const BRIDGE_CHANNEL = 'wxhl-workshop:worldbook';

export interface BridgeRequest {
  channel: typeof BRIDGE_CHANNEL;
  kind: 'request';
  id: string;
  action: string;
  payload: Record<string, any>;
}

export type BridgeResponse =
  | { channel: typeof BRIDGE_CHANNEL; kind: 'response'; id: string; ok: true; payload: unknown }
  | { channel: typeof BRIDGE_CHANNEL; kind: 'response'; id: string; ok: false; error: string };

export function parseBridgeRequest(data: unknown): BridgeRequest | null {
  if (!data || typeof data !== 'object') return null;
  const d = data as Record<string, any>;
  if (d.channel !== BRIDGE_CHANNEL || d.kind !== 'request') return null;
  if (typeof d.id !== 'string' || d.id === '') return null;
  if (typeof d.action !== 'string' || d.action === '') return null;
  return {
    channel: BRIDGE_CHANNEL,
    kind: 'request',
    id: d.id,
    action: d.action,
    payload: (d.payload ?? {}) as Record<string, any>,
  };
}

export function successResponse(id: string, payload: unknown): BridgeResponse {
  return { channel: BRIDGE_CHANNEL, kind: 'response', id, ok: true, payload };
}

export function failureResponse(id: string, error: unknown): BridgeResponse {
  return {
    channel: BRIDGE_CHANNEL,
    kind: 'response',
    id,
    ok: false,
    error: error instanceof Error ? error.message : String(error),
  };
}
```

// src/wxhl-cygf/actions.ts
export const WORKSHOP_WORLDBOOK_NAME = '无限回廊mod';
export const WORKSHOP_EXTRA_KEY = 'wxhlWorkshop';

export interface SourceMeta {
  sourceId: string;
  sourceType?: string;
}

/** 读取条目的工坊把手；无把手或把手残缺返回 null */
export function getWorkshopMeta(entry: WorldbookEntry): (Record<string, any> & SourceMeta) | null {
  const meta = entry?.extra?.[WORKSHOP_EXTRA_KEY];
  if (!meta || typeof meta.sourceId !== 'string' || meta.sourceId === '') return null;
  return meta;
}

/** 条目是否属于 sourceMeta 指定的内容包 */
export function matchesSource(entry: WorldbookEntry, sourceMeta: SourceMeta): boolean {
  const meta = getWorkshopMeta(entry);
  if (!meta || !sourceMeta?.sourceId) return false;
  if (meta.sourceId !== sourceMeta.sourceId) return false;
  return !sourceMeta.sourceType || (meta.sourceType ?? '') === sourceMeta.sourceType;
}

/**
 * upsert 纯逻辑：删掉 sourceMeta 对应的旧条目 → 追加新条目 → 带把手条目按 order 归一化。
 * 不动无把手条目（玩家手动加的内容），不分配 uid。
 *
 * 返回数组按 `position.order` 排好序（managed 以归一化后的 1..N 顺序排在前列）。
 * 注意：必须 sort `merged` 本身——只 sort filter 出来的临时数组会让返回值仍是无序的。
 */
export function computeUpsertWorldbook(
  entries: WorldbookEntry[],
  sourceMeta: SourceMeta,
  newEntries: WorldbookEntry[],
): WorldbookEntry[] {
  const remaining = entries.filter(entry => !matchesSource(entry, sourceMeta));
  const merged = [...remaining, ...newEntries];
  merged.sort((a, b) => (a.position?.order ?? 0) - (b.position?.order ?? 0) || a.name.localeCompare(b.name));
  const managed = merged.filter(entry => getWorkshopMeta(entry) !== null);
  managed.forEach((entry, index) => {
    entry.position = { ...entry.position, order: index + 1 };
  });
  return merged;
}
```

export const MUTATING_ACTIONS: ReadonlySet<string> = new Set([
  'getContext',
  'createWorldbook',
  'createEntries',
  'updateEntry',
  'deleteEntries',
  'upsertWorkshopEntries',

function requireString(payload: Record<string, any>, key: string): string {
  const value = payload[key];
  if (typeof value !== 'string' || value === '') throw new Error(`缺少必要参数: ${key}`);
  return value;
}

function requireEntries(payload: Record<string, any>): WorldbookEntry[] {
  if (!Array.isArray(payload.entries)) throw new Error('缺少必要参数: entries（数组）');
  return payload.entries as WorldbookEntry[];
}

async function handleAction(action: string, payload: Record<string, any>): Promise<unknown> {
  switch (action) {
    case 'handshake':
      return {
        label: 'wxhl创意工坊宿主',
        workshopWorldbook: WORKSHOP_WORLDBOOK_NAME,
        capabilities: {
          getWorldbookNames: typeof getWorldbookNames === 'function',
          getCharWorldbookNames: typeof getCharWorldbookNames === 'function',
          rebindCharWorldbooks: typeof rebindCharWorldbooks === 'function',
          getWorldbook: typeof getWorldbook === 'function',
          createWorldbook: typeof createWorldbook === 'function',
          replaceWorldbook: typeof replaceWorldbook === 'function',
          updateWorldbookWith: typeof updateWorldbookWith === 'function',
          createWorldbookEntries: typeof createWorldbookEntries === 'function',
          deleteWorldbookEntries: typeof deleteWorldbookEntries === 'function',
        },
      };
    case 'getContext': {
      const ensured = await ensureWorkshopWorldbook();
      return {
        worldbookNames: getWorldbookNames(),
        charWorldbooks: getCharWorldbookNames('current'),
        workshopWorldbook: WORKSHOP_WORLDBOOK_NAME,
        ensured,
      };
    }
    // 先校验参数、再引用全局：全局缺失时 ReferenceError 会先于参数校验抛出，掩盖缺参错误
    case 'getEntries': {
      const name = requireString(payload, 'worldbookName');
      return { entries: await getWorldbook(name) };
    }
    case 'createWorldbook': {
      const name = requireString(payload, 'worldbookName');
      return { created: await createWorldbook(name, []) };
    }
    case 'createEntries': {
      const name = requireString(payload, 'worldbookName');
      const entries = requireEntries(payload);
      return await createWorldbookEntries(name, entries, { render: 'immediate' });
    }
    case 'updateEntry': {
      const name = requireString(payload, 'worldbookName');
      const uid = Number(payload.uid);
      if (!Number.isFinite(uid)) throw new Error('updateEntry 缺少合法 uid');
      const nextEntry = (payload.nextEntry ?? {}) as Partial<WorldbookEntry>;
      await updateWorldbookWith(name, worldbook => worldbook.map(entry => (Number(entry.uid) === uid ? { ...entry, ...nextEntry } : entry)), { render: 'immediate' });
      return { updated: true };
    }
    case 'deleteEntries': {
      const name = requireString(payload, 'worldbookName');
      const uids = new Set(((payload.uids ?? []) as unknown[]).map(Number));
      return await deleteWorldbookEntries(name, entry => uids.has(Number(entry.uid)), { render: 'immediate' });
    }
    case 'upsertWorkshopEntries': {
      const name = requireString(payload, 'worldbookName');
      const sourceMeta = payload.sourceMeta as SourceMeta | undefined;
      if (!sourceMeta || typeof sourceMeta.sourceId !== 'string' || sourceMeta.sourceId === '') {
        throw new Error('upsertWorkshopEntries 缺少 sourceMeta.sourceId');
      }
      const newEntries = requireEntries(payload);
      const worldbook = await getWorldbook(name);
      const next = computeUpsertWorldbook(worldbook, sourceMeta, newEntries);
      await replaceWorldbook(name, next, { render: 'immediate' });
      return { total: next.length, installed: newEntries.length };
    }
    default:
      throw new Error(`未知桥接动作: ${action}`);
  }
}

/** 写动作进互斥队列，读动作直接执行 */
export function dispatchAction(action: string, payload: Record<string, any>): Promise<unknown> {
  return MUTATING_ACTIONS.has(action) ? queued(() => handleAction(action, payload)) : handleAction(action, payload);
}

// ==================== 监听器装配 ====================
/** 挂在酒馆主窗口上的运行时键：热重载时用它找到并拆掉上一实例的监听 */
const BRIDGE_RUNTIME_KEY = '__wxhlCygfBridgeTeardown';

export interface BridgeDeps {
  allowedOrigin: string;
  dispatch: (action: string, payload: Record<string, any>) => Promise<unknown>;
}

/**
 * 依赖注入的监听器工厂：真实装配和测试共用同一套校验/分发/回包逻辑。
 * 安全约定：只接受 allowedOrigin 的请求；回包发 event.origin，绝不用 '*'。
 */
export function createBridgeListener(deps: BridgeDeps): (event: MessageEvent) => Promise<void> {
  return async (event: MessageEvent) => {
    const request = parseBridgeRequest(event.data);
    if (!request) return;
    if (event.origin !== deps.allowedOrigin) {
      console.warn(`[wxhl创意工坊] 拒绝来自 ${event.origin} 的桥接消息`);
      return;
    }
    const source = event.source as Window | null;
    try {
      const payload = await deps.dispatch(request.action, request.payload);
      source?.postMessage(successResponse(request.id, payload), event.origin);
    } catch (error) {
      console.error(`[wxhl创意工坊] 桥接动作 ${request.action} 失败:`, error);
      source?.postMessage(failureResponse(request.id, error), event.origin);
    }
  };
}

/** 挂到酒馆主页面窗口（工坊 iframe 的 postMessage 发往其父窗口）。返回拆卸函数。 */
export function installBridge(allowedOrigin: string): () => void {
  const target = window.parent ?? window;
  // 脚本热重载不会触发 pagehide：重载时必须先拆掉上一实例的监听，
  // 否则一条请求会被处理两次（世界书写两遍、同一 id 回两个包）
  const previous = (target as any)[BRIDGE_RUNTIME_KEY];
  if (typeof previous === 'function') previous();

  const listener = createBridgeListener({ allowedOrigin, dispatch: dispatchAction });
  target.addEventListener('message', listener);

  const teardown = () => {
    target.removeEventListener('message', listener);
    if ((target as any)[BRIDGE_RUNTIME_KEY] === teardown) delete (target as any)[BRIDGE_RUNTIME_KEY];
  };
  (target as any)[BRIDGE_RUNTIME_KEY] = teardown;

  console.info('[wxhl创意工坊] 世界书桥接已启用, 允许来源:', allowedOrigin);
  return teardown;
}

      const override = getVariables({ type: 'script', script_id: getScriptId() })?.[URL_OVERRIDE_KEY];

```typescript
// src/wxhl-cygf/discord.ts
import { PANEL_ID, workshopOrigin } from './panel';

export const AUTH_CHANNEL = 'wxhl-workshop:auth';
/** 挂在酒馆主窗口上的运行时键：热重载时用它找到并拆掉上一实例的监听 */
const AUTH_RUNTIME_KEY = '__wxhlCygfAuthTeardown';

export type AuthMessage =
  | { kind: 'open'; url: string }
  | { kind: 'success'; hash: string }
  | { kind: 'error'; error: string; errorDescription?: string };

/** 把原始 postMessage data 归类为登录消息；不认得的一律 null */
export function classifyAuthMessage(data: unknown): AuthMessage | null {
  if (!data || typeof data !== 'object') return null;
  const d = data as Record<string, any>;
  if (d.channel !== AUTH_CHANNEL) return null;
  switch (d.action) {
    case 'openDiscordLogin':
      return typeof d.url === 'string' && d.url !== '' ? { kind: 'open', url: d.url } : null;
    case 'discordLoginSuccess':
      return typeof d.hash === 'string' ? { kind: 'success', hash: d.hash } : null;
    case 'discordLoginError':
      return {
        kind: 'error',
        error: typeof d.error === 'string' && d.error !== '' ? d.error : '未知错误',
        errorDescription: typeof d.errorDescription === 'string' ? d.errorDescription : undefined,
      };
    default:
      return null;
  }
}

/**
 * 登录转发（范例一职责，修正其通配符 origin 缺陷）：
 * 工坊请求开授权弹窗 → 授权完成页通知 opener → 宿主把 hash 转发回工坊 iframe（限定工坊 origin）。
 * 挂在酒馆主页面窗口上；返回拆卸函数。
 */
export function installAuthRelay(): () => void {
  let loginWindow: Window | null = null;
  // 开登录时捕获工坊 iframe 的 window：面板被关掉后 DOM 查不到，但已捕获的引用仍可回传
  let workshopFrame: Window | null = null;

  const getFrameWindow = (): Window | null =>
    ($(`#${PANEL_ID} iframe.wxhl-cygf-frame`)[0] as HTMLIFrameElement | undefined)?.contentWindow ?? null;

  const listener = (event: MessageEvent) => {
    const message = classifyAuthMessage(event.data);
    if (!message) return;
    // 与 bridge.ts 同一条纪律：channel 是公开常量、不构成认证，必须校验来源
    // （云端需把 OAuth 回调页部署在工坊同源下，否则 success/error 会被这里拒掉）
    if (event.origin !== workshopOrigin()) {
      console.warn(`[wxhl创意工坊] 拒绝来自 ${event.origin} 的登录消息`);
      return;
    }
    switch (message.kind) {
      case 'open': {
        workshopFrame = (event.source as Window | null) ?? getFrameWindow();
        if (loginWindow && !loginWindow.closed) loginWindow.close();
        loginWindow = window.parent.open(message.url, 'discord_login', 'width=600,height=800,scrollbars=yes,resizable=yes');
        if (!loginWindow) {
          toastr.warning('请允许弹出窗口以完成 Discord 登录', '创意工坊');
          return;
        }
        toastr.info('请在弹出窗口中完成 Discord 登录', '创意工坊');
        return;
      }
      case 'success': {
        const target = workshopFrame && !workshopFrame.closed ? workshopFrame : getFrameWindow();
        if (!target) {
          // 面板已关：不能既丢 hash 又报成功（静默空壳）
          toastr.warning('登录已完成，但工坊面板已关闭 —— 请重新打开创意工坊', '创意工坊');
        } else {
          target.postMessage({ channel: AUTH_CHANNEL, action: 'discordLoginComplete', hash: message.hash }, workshopOrigin());
          toastr.success('Discord 登录成功', '创意工坊');
        }
        setTimeout(() => {
          if (loginWindow && !loginWindow.closed) loginWindow.close();
        }, 200);
        return;
      }
      case 'error': {
        toastr.error(message.errorDescription ?? message.error, 'Discord 登录失败');
        if (loginWindow && !loginWindow.closed) loginWindow.close();
        return;
      }
    }
  };

  const target = window.parent ?? window;
  // 热重载不触发 pagehide：先拆上一实例的监听，否则一次登录请求会弹两个窗口、成功回执发两遍
  const previous = (target as any)[AUTH_RUNTIME_KEY];
  if (typeof previous === 'function') previous();

  target.addEventListener('message', listener);
  const teardown = () => {
    target.removeEventListener('message', listener);
    if ((target as any)[AUTH_RUNTIME_KEY] === teardown) delete (target as any)[AUTH_RUNTIME_KEY];
  };
  (target as any)[AUTH_RUNTIME_KEY] = teardown;

  console.info('[wxhl创意工坊] Discord 登录转发已启用');
  return teardown;
}
```
